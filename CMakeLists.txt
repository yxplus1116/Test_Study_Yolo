cmake_minimum_required(VERSION 3.24)
project(TestStudyYolo VERSION 2.0 LANGUAGES CXX)
if(MSVC AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "娉ㄦ剰")
  # Repair detection from a cache created under a legacy Chinese code page.
  set(CMAKE_CXX_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${CMAKE_CXX_CL_SHOWINCLUDES_PREFIX}")
endif()
if(NOT WIN32)
  message(FATAL_ERROR "The active desktop application requires Windows. Historical Linux examples are documented in cmake/LegacyCMakeLists.txt.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(LOCAL_DEPS "${CMAKE_SOURCE_DIR}/.local" CACHE PATH "Local dependency root")
set(OpenCV_DIR "${LOCAL_DEPS}/opencv-dist/opencv/build" CACHE PATH "")
find_package(OpenCV 4.10 REQUIRED COMPONENTS core imgproc imgcodecs highgui videoio)
add_library(control_core src/pid.cpp src/mouse.cpp src/makcu_mouse.cpp)
target_include_directories(control_core PUBLIC src src/application)
target_compile_definitions(control_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(control_core PRIVATE /W4 /utf-8)
add_library(vision_core src/runtime/postprocess.cpp src/application/preview.cpp)
target_link_libraries(vision_core PUBLIC control_core ${OpenCV_LIBS})
target_include_directories(vision_core PUBLIC ${OpenCV_INCLUDE_DIRS})
add_executable(TestStudyYolo src/main.cpp src/config.cpp src/dxgi-cap.cpp src/runtime/trt_detector.cpp
  src/application/input_diagnostics.cpp)
target_include_directories(TestStudyYolo SYSTEM PRIVATE "${LOCAL_DEPS}/tensorrt/include" "${LOCAL_DEPS}/cuda/include")
target_link_libraries(TestStudyYolo PRIVATE vision_core
  "${LOCAL_DEPS}/import-libs/nvinfer.lib" "${LOCAL_DEPS}/import-libs/nvonnxparser.lib"
  "${LOCAL_DEPS}/cuda/lib/x64/cudart.lib" d3d11 dxgi user32 gdi32 bcrypt advapi32)
target_compile_options(TestStudyYolo PRIVATE /W4 /utf-8)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE vision_core)
target_compile_options(core_tests PRIVATE /W4 /utf-8)
add_test(NAME core_regressions COMMAND core_tests)
set_tests_properties(core_regressions PROPERTIES TIMEOUT 30)
install(TARGETS TestStudyYolo RUNTIME DESTINATION .)
